# 028-lygia

Here are examples of using the [Lygia Shader Library](https://lygia.xyz/) with [IRMF](https://irmf.io/).

## lygia-01.irmf

![lygia-01.png](lygia-01.png)

**GLSL**

```glsl
/*{
  irmf: "1.0",
  language: "glsl",
  materials: ["jasper","sapphire","agate","emerald"],
  max: [3,3,3],
  min: [-3,-3,-3],
  units: "mm",
}*/

#define FNC_SATURATE
#include "lygia/sdf.glsl"

float sdf2mat(in float a) {
  return smoothstep(0.010, 0.009, a);
}

vec4 sdfs(in vec3 pos) {
  vec4 res = vec4(0);
  
  res.x += sdf2mat(sphereSDF(pos - vec3(0.0, 0.60, 0.0), 0.5));
  res.y += sdf2mat(boxSDF(pos - vec3(2.0, 0.5, 0.0), vec3(0.4)));
  res.z += sdf2mat(torusSDF(pos - vec3(0.0, 0.5, 2.0), vec2(0.4, 0.1)));
  res.w += sdf2mat(capsuleSDF(pos, vec3(-2.3, 0.4, - 0.2), vec3(-1.6, 0.75, 0.2), 0.2));
  res.x += sdf2mat(triPrismSDF(pos - vec3(-2.0, 0.50, - 2.0), vec2(0.5, 0.1)));
  res.y += sdf2mat(cylinderSDF(pos - vec3(2.0, 0.50, - 2.0), vec2(0.2, 0.4)));
  res.z += sdf2mat(coneSDF(pos - vec3(0.0, 0.75, - 2.0), vec3(0.8, 0.6, 0.6)));
  res.w += sdf2mat(hexPrismSDF(pos - vec3(-2.0, 0.60, 2.0), vec2(0.5, 0.1)));
  res.x += sdf2mat(pyramidSDF(pos - vec3(2.0, 0.10, 2.0), 1.0));
  
  return res;
}

void mainModel4(out vec4 materials, in vec3 xyz) {
  materials = sdfs(xyz);
}
```

**TSL**

```tsl
/*{
  irmf: "1.0",
  language: "tsl",
  tslVersion: "0.180.0",
  materials: ["jasper","sapphire","agate","emerald"],
  max: [3,3,3],
  min: [-3,-3,-3],
  units: "mm",
}*/

// Inlined from LYGIA's sdf.glsl (Inigo Quilez's MIT-licensed SDFs).
// `saturate` is provided by three/tsl.
const sdf2mat = (a) => smoothstep(0.010, 0.009, a);

const sphereSDF = (p, s) => length(p).sub(s);

const boxSDF = (p, b) => {
  const d = p.abs().sub(b);
  return min(max(d.x, max(d.y, d.z)), 0.0).add(length(max(d, vec3(0, 0, 0))));
};

const torusSDF = (p, t) => {
  const q = vec2(length(p.xz).sub(t.x), p.y);
  return length(q).sub(t.y);
};

const capsuleSDF = (p, a, b, r) => {
  const pa = p.sub(a);
  const ba = b.sub(a);
  const h = saturate(dot(pa, ba).div(dot(ba, ba)));
  return length(pa.sub(ba.mul(h))).sub(r);
};

const triPrismSDF = (p, h) => {
  const q = p.abs();
  return max(
    q.z.sub(h.y),
    max(q.x.mul(0.866025).add(p.y.mul(0.5)), p.y.negate()).sub(h.x.mul(0.5))
  );
};

const cylinderSDF = (p, h) => {
  const d = vec2(length(p.xz), p.y).abs().sub(h);
  return min(max(d.x, d.y), 0.0).add(length(max(d, vec2(0, 0))));
};

const coneSDF = (p, c) => {
  const q = vec2(length(p.xz), p.y);
  const d1 = q.y.negate().sub(c.z);
  const d2 = max(dot(q, c.xy), q.y);
  return length(max(vec2(d1, d2), vec2(0, 0))).add(min(max(d1, d2), 0.0));
};

const hexPrismSDF = (p, h) => {
  const q = p.abs();
  const d1 = q.z.sub(h.y);
  const d2 = max(q.x.mul(0.866025).add(q.y.mul(0.5)), q.y).sub(h.x);
  return length(max(vec2(d1, d2), vec2(0, 0))).add(min(max(d1, d2), 0.0));
};

const pyramidSDF = (p, h) => {
  const m2 = h * h + 0.25;
  let xz = p.xz.abs();
  xz = select(xz.y.greaterThan(xz.x), vec2(xz.y, xz.x), xz);
  xz = xz.sub(vec2(0.5, 0.5));
  const q = vec3(
    xz.y,
    float(h).mul(p.y).sub(xz.x.mul(0.5)),
    float(h).mul(xz.x).add(p.y.mul(0.5))
  );
  const s = max(q.x.negate(), 0.0);
  const t = saturate(q.y.sub(xz.y.mul(0.5)).div(m2 + 0.25));
  const a = float(m2).mul(q.x.add(s)).mul(q.x.add(s)).add(q.y.mul(q.y));
  const qy2 = q.y.sub(t.mul(m2));
  const b = float(m2).mul(q.x.add(t.mul(0.5))).mul(q.x.add(t.mul(0.5))).add(qy2.mul(qy2));
  const inside = min(q.y, q.x.negate().mul(m2).sub(q.y.mul(0.5))).greaterThan(0.0);
  const d2 = select(inside, float(0.0), min(a, b));
  return sqrt(d2.add(q.z.mul(q.z)).div(m2)).mul(sign(max(q.z, p.y.negate())));
};

const sdfs = (pos) => {
  let rx = float(0.0);
  let ry = float(0.0);
  let rz = float(0.0);
  let rw = float(0.0);

  rx = rx.add(sdf2mat(sphereSDF(pos.sub(vec3(0.0, 0.60, 0.0)), 0.5)));
  ry = ry.add(sdf2mat(boxSDF(pos.sub(vec3(2.0, 0.5, 0.0)), vec3(0.4))));
  rz = rz.add(sdf2mat(torusSDF(pos.sub(vec3(0.0, 0.5, 2.0)), vec2(0.4, 0.1))));
  rw = rw.add(sdf2mat(capsuleSDF(
    pos,
    vec3(-2.3, 0.4, -0.2),
    vec3(-1.6, 0.75, 0.2),
    0.2
  )));
  rx = rx.add(sdf2mat(triPrismSDF(pos.sub(vec3(-2.0, 0.50, -2.0)), vec2(0.5, 0.1))));
  ry = ry.add(sdf2mat(cylinderSDF(pos.sub(vec3(2.0, 0.50, -2.0)), vec2(0.2, 0.4))));
  rz = rz.add(sdf2mat(coneSDF(pos.sub(vec3(0.0, 0.75, -2.0)), vec3(0.8, 0.6, 0.6))));
  rw = rw.add(sdf2mat(hexPrismSDF(pos.sub(vec3(-2.0, 0.60, 2.0)), vec2(0.5, 0.1))));
  rx = rx.add(sdf2mat(pyramidSDF(pos.sub(vec3(2.0, 0.10, 2.0)), 1.0)));

  return [rx, ry, rz, rw];
};

const mainModel4 = (xyz) => sdfs(xyz);
```
