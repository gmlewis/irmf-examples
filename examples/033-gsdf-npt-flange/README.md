const gsdfWinding = (p, v1, v2, ds) => {
  const e = v2.sub(v1);
  const w = p.sub(v1);
  const b = w.sub(e.mul(clamp(dot(w, e).div(dot(e, e)), 0.0, 1.0)));
  const dmin = min(ds.x, dot(b, b));
  const c0 = p.y.greaterThanEqual(v1.y);
  const c1 = p.y.lessThan(v2.y);
  const c2 = e.x.mul(w.y).greaterThan(e.y.mul(w.x));
  const flip = c0.and(c1).and(c2).or(c0.not().and(c1.not()).and(c2.not()));
  return vec2(dmin, select(flip, ds.y.negate(), ds.y));
};

const makePoly = (verts) => (p) => {
  const v = verts.map(([x, y]) => vec2(x, y));
  const num = v.length;
  let ds = vec2(dot(p.sub(v[0]), p.sub(v[0])), float(1.0));
  for (let i = 0, j = num - 1; i < num; j = i, i++) {
    ds = gsdfWinding(p, v[i], v[j], ds);
  }
  return ds.y.mul(sqrt(ds.x));
};

const screw = (p) => {
  const lead = -0.071428575;
  const pitch = 0.071428575;
  const taper = 0.031239834;
  const L = 0.214271203;
  let y = length(p.xy);
  if (taper !== 0) {
    y = y.add(p.z.mul(Math.atan(taper)));
  }
  const theta = atan(p.y, p.x);
  const z = p.z.add(theta.mul(lead).div(2.0 * Math.PI));
  const sawt = z.add(pitch / 2.0).div(pitch);
  const saw = float(pitch).mul(sawt.sub(floor(sawt))).sub(0.5 * pitch);
  const p2 = vec2(saw, y);
  const d2 = poly2D(p2);
  const d3 = p.z.abs().sub(L);
  return max(d2, d3);
};
